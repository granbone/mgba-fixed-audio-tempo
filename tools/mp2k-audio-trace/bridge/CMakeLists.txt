cmake_minimum_required(VERSION 3.20)
project(mgba_mp2k_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT AGBPLAY_SOURCE_DIR)
  message(FATAL_ERROR "Set AGBPLAY_SOURCE_DIR to the external ipatix/agbplay checkout")
endif()
set(AGB ${AGBPLAY_SOURCE_DIR}/src/agbplay)
set(AGB_SOURCES
  CGBPatterns.cpp Debug.cpp FileReader.cpp Gsf.cpp LoudnessCalculator.cpp
  MP2KChn.cpp MP2KChnPCM.cpp MP2KChnPSG.cpp MP2KContext.cpp MP2KPlayer.cpp
  MP2KScanner.cpp MP2KTrack.cpp Resampler.cpp ResamplerAVX2.cpp
  ReverbEffect.cpp Rom.cpp SequenceReader.cpp SoundMixer.cpp Types.cpp Xcept.cpp)
list(TRANSFORM AGB_SOURCES PREPEND "${AGB}/")
option(MP2K_PORTABLE_STATIC_DEPS "Link portable Windows dependencies statically" OFF)
if(MP2K_RELINK_SUPPORT_DIR)
  if(NOT MP2K_PORTABLE_STATIC_DEPS)
    message(FATAL_ERROR "Relinking support requires portable static dependencies")
  endif()
  set(MP2K_STATIC_LIB_DIR "${MP2K_RELINK_SUPPORT_DIR}/lib")
  set(FMT_HEADERS "${MP2K_RELINK_SUPPORT_DIR}/include")
else()
  find_package(fmt REQUIRED)
  find_package(Boost REQUIRED)
  find_package(libzip REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(ZLIB REQUIRED IMPORTED_TARGET zlib)
endif()
add_library(agbplay_candidate OBJECT ${AGB_SOURCES})
set_source_files_properties(${AGB}/ResamplerAVX2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2")
target_include_directories(agbplay_candidate PUBLIC ${AGB})
if(NOT MP2K_RELINK_SUPPORT_DIR)
  target_link_libraries(agbplay_candidate PRIVATE Boost::boost)
endif()
if(MP2K_PORTABLE_STATIC_DEPS)
  target_compile_definitions(agbplay_candidate PRIVATE ZIP_STATIC)
  if(NOT MP2K_RELINK_SUPPORT_DIR)
    get_target_property(FMT_HEADERS fmt::fmt INTERFACE_INCLUDE_DIRECTORIES)
  endif()
  target_include_directories(agbplay_candidate PRIVATE ${FMT_HEADERS})
else()
  target_link_libraries(agbplay_candidate PRIVATE fmt::fmt)
endif()
add_library(mgba_mp2k_bridge SHARED bridge.cpp $<TARGET_OBJECTS:agbplay_candidate>)
target_include_directories(mgba_mp2k_bridge PRIVATE ${AGB})
if(MP2K_RELINK_SUPPORT_DIR)
  target_include_directories(mgba_mp2k_bridge PRIVATE "${MP2K_RELINK_SUPPORT_DIR}/include")
endif()
if(MP2K_PORTABLE_STATIC_DEPS)
  if(NOT MINGW OR NOT MP2K_STATIC_LIB_DIR)
    message(FATAL_ERROR "Portable static dependencies require MinGW and MP2K_STATIC_LIB_DIR")
  endif()
  # This changes packaging only. The validated player and C ABI are unchanged.
  foreach(name fmt zip bz2 lzma zstd z)
    set(archive "${MP2K_STATIC_LIB_DIR}/lib${name}.a")
    if(NOT EXISTS "${archive}")
      message(FATAL_ERROR "Missing static dependency: ${archive}")
    endif()
    target_link_libraries(mgba_mp2k_bridge PRIVATE "${archive}")
  endforeach()
  target_link_libraries(mgba_mp2k_bridge PRIVATE bcrypt advapi32)
else()
  target_link_libraries(mgba_mp2k_bridge PRIVATE fmt::fmt libzip::zip PkgConfig::ZLIB)
endif()
if(MINGW)
  target_link_options(mgba_mp2k_bridge PRIVATE -static)
endif()
